#include "Framebuffer.h"
#include "Texture2D.h"
// #include "Renderbuffer.h" // 如果使用 Renderbuffer 深度/模板

// ... 在某个初始化函数中 ...

int screenWidth = 1280;
int screenHeight = 720;

// 1. 创建 FBO 实例
auto gBufferFBO = std::make_unique<Framebuffer>(screenWidth, screenHeight);

// 2. 创建颜色附件纹理 (Texture2D 类管理)
auto albedoTexture = std::make_unique<Texture2D>(screenWidth, screenHeight, GL_RGBA8);
auto normalTexture = std::make_unique<Texture2D>(screenWidth, screenHeight, GL_RGB16F);
auto positionTexture = std::make_unique<Texture2D>(screenWidth, screenHeight, GL_RGBA16F); // 例如，存储世界空间位置

// 3. 创建深度附件纹理 (Texture2D 类管理)
auto depthTexture = std::make_unique<Texture2D>(screenWidth, screenHeight, GL_DEPTH24_STENCIL8); // 或者使用 Renderbuffer

// 4. 将纹理附加到 FBO
gBufferFBO->attachColorTexture(albedoTexture->id(), GL_COLOR_ATTACHMENT0);
gBufferFBO->attachColorTexture(normalTexture->id(), GL_COLOR_ATTACHMENT1);
gBufferFBO->attachColorTexture(positionTexture->id(), GL_COLOR_ATTACHMENT2);
gBufferFBO->attachDepthTexture(depthTexture->id()); // 或者 attachDepthStencilTexture

// 5. 设置绘制缓冲区
std::vector<GLenum> drawBuffers = {GL_COLOR_ATTACHMENT0, GL_COLOR_ATTACHMENT1, GL_COLOR_ATTACHMENT2};
gBufferFBO->setDrawBuffers(drawBuffers);

// 6. 检查 FBO 完整性
gBufferFBO->checkCompleteness();

// 现在 gBufferFBO 已准备好使用，并且 albedoTexture, normalTexture, positionTexture, depthTexture
